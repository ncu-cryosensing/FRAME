# FRAME-formatted JSON example
This example represent the metadata JSON structure in the FRAME format. The placeholder values ("...") should be replaced with the corresponding metadata values.
``` json

{
  "metadataGeneratedBy": "...",
  "id": "...",
  "repository_name": "...",
  "title": "...",
  "metadataIdentifier": "...",
  "metadataCode": "...",
  "authors": [
    {
      "name": "...",
      "orcid": "...",
      "affiliation": "...",
      "aff_numb": "..."
    }
  ],
  "authors_email": "...",
  "publicationDate": "...",
  "doi": "...",
  "short_description": "...",
  "license": "...",
  "documentation": "...",
  "url_page": "...",
  "url_download": "...",
  "url_map": "...",
  "cloud_environment": "...",
  "url_api": "...",
  "spatialExtent": {
    "...": "..."
  },
  "fundername": "...",
  "award": "...",
  "resourceType": "..."
}

```
